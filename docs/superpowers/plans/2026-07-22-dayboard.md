# dayboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 달력·날씨·해/달·천체를 한 화면씩 넘겨 보는 전체화면 스와이프 웹 표출 보드(uv+Flask+waitress), server-manager가 관리하는 서비스 계약을 그대로 따르는 재사용 test fixture.

**Architecture:** schedule-dashboard 구조 미러링 — `src/dayboard/`에 순수 데이터 모듈(weather/astro/calendar_data)과 `server/`(Flask app factory·access-log 미들웨어·blueprints). 프론트는 `src/dayboard/static/`의 vanilla JS/CSS(빌드 0). 백엔드는 `/api/data`로 4개 pane 데이터를 한 JSON에 조립, 프론트가 폴링. 네트워크는 weather(Open-Meteo)만, 나머지는 오프라인.

**Tech Stack:** Python 3.14, uv, hatchling, Flask, waitress, requests, astral 3.2, holidays 0.101, pytest. Frontend: vanilla HTML/JS/CSS.

## Global Constraints

- **Python 3.14** (`.python-version`), **uv** 패키지 관리, **hatchling** 빌드. deps: `flask`, `waitress`, `requests`, `astral>=3.2`, `holidays>=0.101`. dev: `pytest`.
- **secret 0**: API 키·비밀 파일 절대 금지. **빌드 0**: 프론트 빌드 스텝 없음(vanilla).
- **날씨 = Open-Meteo best_match** (실측 확정): `GET https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m&daily=temperature_2m_max,temperature_2m_min,weather_code&timezone={tz}&forecast_days=1`. **`/v1/kma`(404)·`models=kma_seamless`(null 반환) 쓰지 말 것** — 기본 best_match가 한국에서 KMA 고해상 모델을 자동 선택+폴백한다. (spec §11-1 해소.)
- **astral 3.2 API** (실측): `from astral import LocationInfo; from astral.sun import sun; from astral.moon import phase`. `sun(loc.observer, date=d, tzinfo="Asia/Seoul")` → `{"dawn","sunrise","noon","sunset","dusk"}` (tz-aware datetime). `phase(d)` → `float` 0..27.99 (달 나이). (spec §11-2 해소.)
- **holidays 0.101 API** (실측): `holidays.SouthKorea(years=y, language="ko")`; `.get(date)` → 한국어 이름 or None; `date in kr` → bool. (spec §11-3 해소.)
- **access-log 형식** (schedule-dashboard 복제): `<ip> "<method> <path>" <status>`, REMOTE_ADDR 없으면 `-`.
- **계약**: `server.toml`에 `[hooks] post_install = "bash setup/install.sh"`; `[[service]]` port **20000**, health_path `/health`, autostart_service `dayboard`, pidfile `data/dayboard.pid`. systemd 유닛 `dayboard.service`(ExecStart=`bash setup/run.sh`, WorkingDirectory 줄에 인라인 `#` 금지). 위치 기본값 서울(37.57, 126.98, "서울"), timezone Asia/Seoul, port 20000, weather cache TTL 600s는 `config.toml`.
- **커밋 케이던스**: implementer는 commit 안 함 → controller가 working-tree diff 리뷰 → 사용자가 task당 직접 add+commit. commit step 메시지는 제안값. **브랜치 = main**(그린필드, spec 커밋만 있음 → 신규 구현은 main에 직접).
- 모든 Python은 `cwd` 비의존(파일 기준 절대경로). 텍스트 `open()`은 `encoding="utf-8"`.

```python
"""설정 SSOT: config.toml 로드 + 경로 상수. cwd 비의존."""
import tomllib
from dataclasses import dataclass
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parents[2]   # src/dayboard/config.py -> repo root
CONFIG_TOML = PROJECT_ROOT / "config.toml"
DATA_DIR = PROJECT_ROOT / "data"


@dataclass(frozen=True)
class Config:
    lat: float
    lon: float
    label: str
    tz: str
    port: int
    cache_ttl: int


def load() -> Config:
    loc, srv, wx = {}, {}, {}
    if CONFIG_TOML.exists():
        data = tomllib.loads(CONFIG_TOML.read_text(encoding="utf-8"))
        loc, srv, wx = data.get("location", {}), data.get("server", {}), data.get("weather", {})
    return Config(
        lat=float(loc.get("lat", 37.57)),
        lon=float(loc.get("lon", 126.98)),
        label=str(loc.get("label", "서울")),
        tz=str(loc.get("tz", "Asia/Seoul")),
        port=int(srv.get("port", 20000)),
        cache_ttl=int(wx.get("cache_ttl", 600)),
    )


def ensure_runtime_dirs() -> None:
    DATA_DIR.mkdir(parents=True, exist_ok=True)
```

```python
"""이번달 그리드 + 한국 공휴일(오프라인)."""
import calendar
import datetime

import holidays


def month_grid(year: int, month: int, today: datetime.date) -> dict:
    kr = holidays.SouthKorea(years=year, language="ko")
    cal = calendar.Calendar(firstweekday=0)  # 0 = 월요일
    weeks = []
    for week in cal.monthdayscalendar(year, month):  # 0 = 빈칸
        row = []
        for wd, day in enumerate(week):
            if day == 0:
                row.append({"day": None, "is_today": False, "holiday": None, "weekday": wd})
            else:
                d = datetime.date(year, month, day)
                row.append({
                    "day": day,
                    "is_today": d == today,
                    "holiday": kr.get(d),
                    "weekday": wd,
                })
        weeks.append(row)
    return {"year": year, "month": month, "weeks": weeks}
```

```python
"""해·달(astral, 오프라인) + 큐레이션 천체 이벤트."""
import datetime
import json
from pathlib import Path

from astral import LocationInfo
from astral.moon import phase as _moon_phase
from astral.sun import sun as _sun

_EVENTS_PATH = Path(__file__).resolve().parent / "data" / "celestial-events-2026.json"

# 달 나이(0..27.99, 삭망월 ~29.53의 astral 스케일) → 한국어 위상명
_PHASE_BANDS = [
    (1.84, "신월"), (5.53, "초승달"), (9.22, "상현"), (12.91, "상현망간"),
    (16.61, "보름"), (20.30, "하현망간"), (23.99, "하현"), (27.68, "그믐달"),
]


def moon_phase_name(age: float) -> str:
    for upper, name in _PHASE_BANDS:
        if age < upper:
            return name
    return "신월"  # 27.68.. → 다시 신월


def sun_moon(lat: float, lon: float, tz: str, date: datetime.date) -> dict:
    loc = LocationInfo(latitude=lat, longitude=lon, timezone=tz)
    s = _sun(loc.observer, date=date, tzinfo=tz)
    age = float(_moon_phase(date))
    return {
        "sunrise": s["sunrise"].strftime("%H:%M"),
        "sunset": s["sunset"].strftime("%H:%M"),
        "moon_age": round(age, 2),
        "moon_phase": moon_phase_name(age),
    }


def upcoming_events(today: datetime.date, n: int = 5, path: Path | None = None) -> list[dict]:
    p = path or _EVENTS_PATH
    try:
        items = json.loads(p.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return []
    upcoming = [e for e in items if e.get("date", "") >= today.isoformat()]
    upcoming.sort(key=lambda e: e["date"])
    return upcoming[:n]
```

```python
"""Open-Meteo(best_match) 날씨 + 인메모리 TTL 캐시. 네트워크는 여기에만."""
import time

import requests

WMO_LABELS = {
    0: "맑음", 1: "대체로 맑음", 2: "부분적 흐림", 3: "흐림",
    45: "안개", 48: "짙은 안개",
    51: "약한 이슬비", 53: "이슬비", 55: "강한 이슬비",
    61: "약한 비", 63: "비", 65: "강한 비",
    71: "약한 눈", 73: "눈", 75: "강한 눈",
    80: "약한 소나기", 81: "소나기", 82: "강한 소나기",
    95: "뇌우", 96: "우박 동반 뇌우", 99: "강한 우박 뇌우",
}

_URL = "https://api.open-meteo.com/v1/forecast"
_cache: dict | None = None
_cache_at: float = 0.0


def reset_cache() -> None:
    global _cache, _cache_at
    _cache, _cache_at = None, 0.0


def _fetch(lat: float, lon: float, tz: str) -> dict:
    params = {
        "latitude": lat, "longitude": lon,
        "current": "temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m",
        "daily": "temperature_2m_max,temperature_2m_min,weather_code",
        "timezone": tz, "forecast_days": 1,
    }
    resp = requests.get(_URL, params=params, timeout=8)
    resp.raise_for_status()
    return resp.json()


def _parse(payload: dict) -> dict:
    cur, daily = payload["current"], payload["daily"]
    code = int(cur["weather_code"])
    return {
        "temp": cur["temperature_2m"],
        "humidity": cur["relative_humidity_2m"],
        "wind": cur["wind_speed_10m"],
        "code": code,
        "label": WMO_LABELS.get(code, "정보 없음"),
        "tmax": daily["temperature_2m_max"][0],
        "tmin": daily["temperature_2m_min"][0],
    }


def get_weather(lat: float, lon: float, tz: str, ttl: int = 600, now: float | None = None) -> dict | None:
    global _cache, _cache_at
    t = time.time() if now is None else now
    if _cache is not None and (t - _cache_at) < ttl:
        return _cache
    try:
        parsed = _parse(_fetch(lat, lon, tz))
        _cache, _cache_at = parsed, t
        return parsed
    except Exception:
        return _cache  # 실패 → 마지막 캐시(없으면 None)
```

```python
"""요청당 access log 1줄 stdout emit (NCCP: `<ip> "<method> <path>" <status>`)."""


class AccessLog:
    def __init__(self, app):
        self.app = app

    def __call__(self, environ, start_response):
        ip = environ.get("REMOTE_ADDR", "-")
        method = environ.get("REQUEST_METHOD", "-")
        path = environ.get("PATH_INFO", "-")

        def _sr(status, headers, exc_info=None):
            print(f'{ip} "{method} {path}" {status.split(" ", 1)[0]}', flush=True)
            return start_response(status, headers, exc_info)

        return self.app(environ, _sr)
```

```python
"""웹 서버 진입점: server.toml에서 host/port 읽고 waitress로 서빙."""
import argparse
import tomllib

import waitress

from . import config
from .server.access_log import AccessLog
from .server.app import create_app


def _load_web_service():
    c = config.load()
    host, port = "0.0.0.0", c.port
    pidfile = config.DATA_DIR / "dayboard.pid"
    server_toml = config.PROJECT_ROOT / "server.toml"
    if server_toml.exists():
        data = tomllib.loads(server_toml.read_text(encoding="utf-8"))
        for svc in data.get("service", []):
            if svc.get("type") == "web":
                host = svc.get("host", host)
                port = int(svc.get("port", port))
                pidfile = config.PROJECT_ROOT / svc.get("pidfile", "data/dayboard.pid")
                break
    return host, port, pidfile


def main(argv=None):
    h, p, pidfile = _load_web_service()
    ap = argparse.ArgumentParser()
    ap.add_argument("--host", default=h)
    ap.add_argument("--port", type=int, default=p)
    args = ap.parse_args(argv)

    config.ensure_runtime_dirs()
    pidfile.write_text(str(__import__("os").getpid()), encoding="utf-8")
    app = create_app()
    print(f"dayboard serving on http://{args.host}:{args.port}", flush=True)
    waitress.serve(AccessLog(app), host=args.host, port=args.port)


if __name__ == "__main__":
    main()
```

```javascript
// dayboard 프론트: /api/data 폴링 + 4-page 클릭드래그 스와이프.
const PAGES = 4, REFRESH_MS = 5 * 60 * 1000;
let idx = 0;
const track = document.getElementById('track');
const dotsBox = document.getElementById('dots');

for (let i = 0; i < PAGES; i++) {
  const d = document.createElement('div');
  d.className = 'dot' + (i === 0 ? ' on' : '');
  d.onclick = () => go(i);
  dotsBox.appendChild(d);
}
function go(i) {
  idx = Math.max(0, Math.min(PAGES - 1, i));
  track.style.transform = `translateX(-${idx * 25}%)`;
  [...dotsBox.children].forEach((d, k) => d.classList.toggle('on', k === idx));
}
document.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowLeft') go(idx - 1);
  if (e.key === 'ArrowRight') go(idx + 1);
});
// 클릭-드래그 / 터치 스와이프
let x0 = null;
const start = (x) => { x0 = x; track.style.transition = 'none'; };
const end = (x) => {
  track.style.transition = '';
  if (x0 === null) return;
  const dx = x - x0; x0 = null;
  if (Math.abs(dx) > 60) go(idx + (dx < 0 ? 1 : -1)); else go(idx);
};
track.addEventListener('mousedown', (e) => start(e.clientX));
window.addEventListener('mouseup', (e) => end(e.clientX));
track.addEventListener('touchstart', (e) => start(e.touches[0].clientX), { passive: true });
track.addEventListener('touchend', (e) => end(e.changedTouches[0].clientX));

const WD = ['월', '화', '수', '목', '금', '토', '일'];
function renderCalendar(c) {
  let h = '<div class="cal-grid">' + WD.map(w => `<div class="cal-h">${w}</div>`).join('');
  for (const wk of c.weeks) for (const cell of wk) {
    if (cell.day === null) { h += '<div class="cal-c empty"></div>'; continue; }
    const cls = 'cal-c' + (cell.is_today ? ' today' : '') + (cell.holiday ? ' holiday' : '');
    h += `<div class="${cls}">${cell.day}` +
         (cell.holiday ? `<span class="cal-hol-name">${cell.holiday}</span>` : '') + '</div>';
  }
  document.getElementById('cal').innerHTML = h + '</div>';
}
function renderWeather(w) {
  const el = document.getElementById('wx');
  if (!w) { el.innerHTML = '<div class="err">날씨 불러오기 실패</div>'; return; }
  el.innerHTML = `<div class="big">${Math.round(w.temp)}°</div>` +
    `<div class="sub">${w.label}</div>` +
    `<div class="row"><span>최고 ${Math.round(w.tmax)}°</span><span>최저 ${Math.round(w.tmin)}°</span>` +
    `<span>습도 ${w.humidity}%</span><span>바람 ${w.wind} m/s</span></div>`;
}
function renderSunMoon(s) {
  document.getElementById('sm').innerHTML =
    `<div class="row" style="font-size:2rem"><span>🌅 ${s.sunrise}</span><span>🌇 ${s.sunset}</span></div>` +
    `<div class="big" style="font-size:3rem;margin-top:3vh">${s.moon_phase}</div>` +
    `<div class="sub">달 나이 ${s.moon_age}일</div>`;
}
function renderSky(events) {
  const el = document.getElementById('sky');
  if (!events || !events.length) { el.innerHTML = '<div class="err">다가오는 천체 이벤트 없음</div>'; return; }
  el.innerHTML = events.map(e =>
    `<div class="ev"><span>${e.name}${e.note ? ` <small>(${e.note})</small>` : ''}</span><span class="d">${e.date}</span></div>`
  ).join('');
}
async function refresh() {
  try {
    const d = await (await fetch('/api/data')).json();
    renderCalendar(d.calendar); renderWeather(d.weather);
    renderSunMoon(d.sun_moon); renderSky(d.events);
  } catch (e) { /* 네트워크 순간 실패 — 다음 폴링에서 복구 */ }
}
refresh();
setInterval(refresh, REFRESH_MS);
```
